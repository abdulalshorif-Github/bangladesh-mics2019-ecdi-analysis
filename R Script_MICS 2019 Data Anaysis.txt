# ---- 2. Recode variables ------------------------------------------
analysis_data <- mics %>%
  transmute(
    PSU            = factor(PSU),
    STRATUM        = factor(STRATUM),
    survey_weight  = to_numeric(WEIGHT),                 # child/ECDI weight
    ECDI_TRACK     = to_numeric(ECDI_TRACK),
    ecoli          = factor(to_numeric(HH_DW_ECOLI), levels = c(0, 1),

    AGE_M          = to_numeric(AGE_M),
    AGE_M_CENTERED = AGE_M - 48,                          # centered at 48 months
    sex            = factor(to_numeric(SEX), levels = c(1, 2),
                            labels = c("Male", "Female")),
    mother_edu     = factor(to_numeric(MOTHER_EDU), levels = 0:3,
                            labels = c("Pre-primary or none", "Primary",
                                       "Secondary", "Higher secondary+")),
    wealth_score   = to_numeric(HH_WEALTH_SCORE),
    area           = factor(to_numeric(AREA), levels = c(1, 2),
                            labels = c("Urban", "Rural")),
    division       = factor(to_numeric(DIVISION),

                            labels = c("Barishal", "Chattogram", "Dhaka",

  ) %>%
  filter(AGE_M >= 36, AGE_M <= 59) %>%                    # official ECDI age range
  filter(complete.cases(.), survey_weight > 0, ECDI_TRACK %in% c(0, 1))

cat("Final analytical sample:", nrow(analysis_data), "children\n")
stopifnot(all(unique(analysis_data$ECDI_TRACK) %in% c(0, 1)),
          length(unique(analysis_data$ECDI_TRACK)) == 2)

# ---- 3. Survey design ----------------------------------------------
mics_design <- svydesign(ids = ~PSU, strata = ~STRATUM,
                         weights = ~survey_weight, data = analysis_data,
                         nest = TRUE)

# ---- 4. Models -------------------------------------------------------
model0 <- svyglm(ECDI_TRACK ~ ecoli + arsenic,
                 design = mics_design, family = quasibinomial())

model1 <- svyglm(ECDI_TRACK ~ ecoli + arsenic + AGE_M + sex,
                 design = mics_design, family = quasibinomial())

model2 <- svyglm(ECDI_TRACK ~ ecoli + arsenic + AGE_M + sex +
                   mother_edu + wealth_score + area + division,
                 design = mics_design, family = quasibinomial())

model3 <- svyglm(ECDI_TRACK ~ ecoli * arsenic + AGE_M + sex +
                   mother_edu + wealth_score + area + division,
                 design = mics_design, family = quasibinomial())

# ---- 5. Odds-ratio table helper --------------------------------------
or_table <- function(model, name) {
  b <- coef(summary(model))
  ci <- confint(model)
  data.frame(Model = name, Variable = rownames(b),
             Beta = round(b[, "Estimate"], 3),
             Adjusted_OR = round(exp(b[, "Estimate"]), 3),
             CI_95_Lower = round(exp(ci[, 1]), 3),
             CI_95_Upper = round(exp(ci[, 2]), 3),
             P_value = ifelse(b[, ncol(b)] < 0.001, "<0.001",
                              sprintf("%.3f", b[, ncol(b)])),
             row.names = NULL) %>%
    filter(Variable != "(Intercept)")
}

results <- bind_rows(
  or_table(model0, "Model 0"),
  or_table(model1, "Model 1"),
  or_table(model2, "Model 2"),
  or_table(model3, "Model 3 (interaction)")
)

cat("\n=== Odds ratios, all models ===\n"); print(results)

# Is the interaction itself significant? This is the real test of
# whether arsenic and E. coli combine synergistically/additively.
cat("\n=== Model 3: arsenic x E. coli interaction test ===\n")
print(regTermTest(model3, ~ecoli:arsenic))

# ---- 6. Overall (design-based) significance tests for Model 2 --------
cat("\n=== Model 2: overall Wald tests per variable ===\n")
for (v in c("ecoli", "arsenic", "AGE_M", "sex",
            "mother_edu", "wealth_score", "area", "division")) {
  cat("\n", v, ":\n", sep = "")
  print(regTermTest(model2, as.formula(paste0("~", v))))
}

# ---- 7. Does adding covariates improve the model? ---------------------
cat("\n=== Model comparisons (nested Wald tests) ===\n")
cat("\nModel 0 vs 1:\n"); print(anova(model0, model1, method = "Wald"))
cat("\nModel 1 vs 2:\n"); print(anova(model1, model2, method = "Wald"))
cat("\nModel 2 vs 3 (does the interaction improve fit):\n")
print(anova(model2, model3, method = "Wald"))

# ---- 8. Linearity checks (age, wealth) ---------------------------------
model2_age_sq <- update(model2, . ~ . + I(AGE_M^2))
model2_wealth_sq <- update(model2, . ~ . + I(wealth_score^2))
cat("\n=== Linearity checks ===\n")
cat("Age (quadratic term):\n")
print(regTermTest(model2_age_sq, ~I(AGE_M^2)))
cat("Wealth (quadratic term):\n")
print(regTermTest(model2_wealth_sq, ~I(wealth_score^2)))

# ---- 9. Save outputs -----------------------------------------------------
write.csv(results, "all_models_odds_ratios_updated.csv", row.names = FALSE)

cat("\nDone. Saved: all_models_odds_ratios_updated.csv\n")

# ---- 4. Odds-ratio table helper ---------------------------------------
or_table <- function(model, model_name, domain) {
  b <- coef(summary(model))
  ci <- confint(model)
  data.frame(Domain = domain, Model = model_name, Variable = rownames(b),
             Beta = round(b[, "Estimate"], 3),
             Adjusted_OR = round(exp(b[, "Estimate"]), 3),
             CI_95_Lower = round(exp(ci[, 1]), 3),
             CI_95_Upper = round(exp(ci[, 2]), 3),
             P_value = ifelse(b[, ncol(b)] < 0.001, "<0.001",
                              sprintf("%.3f", b[, ncol(b)])),
             row.names = NULL) %>%
    filter(Variable != "(Intercept)")
}

# ---- 5. Separation checks --------------------------------------------
# A categorical covariate causes "separation" if any of its levels has
# ZERO children in one outcome category (e.g. a division with 0 "not on
# track" children). Fitting logistic regression on such data produces
# fake extreme estimates (huge betas, OR near 0 or in the millions,
# CI = [0,0] or absurdly wide) rather than a real effect. This checks
# for that automatically and drops the offending ADJUSTMENT covariate
# (not the exposures) before fitting Model 2/3.
has_zero_cell <- function(data, outcome, var) {
  tab <- table(data[[var]], data[[outcome]])
  any(tab == 0)
}

# The arsenic x E. coli interaction needs its own check: all four
# combinations of (ecoli, arsenic) must contain BOTH outcome values,
# or the interaction term is not estimable.
interaction_estimable <- function(data, outcome) {
  tab <- table(data$ecoli, data$arsenic, data[[outcome]])
  !any(apply(tab, c(1, 2), function(x) any(x == 0)))
}

# ---- 6. Function: fit Models 0-3 for one outcome variable --------------
fit_domain_models <- function(outcome, data, design) {
  
  adjustment_vars <- c("sex", "mother_edu", "area", "division")  # wealth_score
  # is continuous, so it can't cause this kind of zero-cell separation
  
  zero_cell_vars <- adjustment_vars[sapply(adjustment_vars, has_zero_cell,
                                           data = data, outcome = outcome)]
  covars_safe <- setdiff(adjustment_vars, zero_cell_vars)
  # sex and AGE_M_CENTERED are kept in every model; only mother_edu/area/
  # division (plus wealth_score) get added conditionally in Model 2/3
  extra_covars <- setdiff(covars_safe, "sex")
  extra_str <- paste(c(extra_covars, "wealth_score"), collapse = " + ")
  
  if (length(zero_cell_vars) > 0) {
    cat("  [", outcome, "] dropped from Model 2/3 (zero-cell separation): ",
        paste(zero_cell_vars, collapse = ", "), "\n", sep = "")
  }
  
  sex_ok <- !("sex" %in% zero_cell_vars)   # sex is very unlikely to separate,
  sex_term <- if (sex_ok) "+ sex" else ""  # but checked for completeness
  
  f0 <- as.formula(paste(outcome, "~ ecoli + arsenic"))
  f1 <- as.formula(paste(outcome, "~ ecoli + arsenic + AGE_M", sex_term))
  f2 <- as.formula(paste(outcome, "~ ecoli + arsenic + AGE_M", sex_term,
                         "+", extra_str))
  
  m0 <- svyglm(f0, design = design, family = quasibinomial())
  m1 <- svyglm(f1, design = design, family = quasibinomial())
  m2 <- svyglm(f2, design = design, family = quasibinomial())
  
  if (interaction_estimable(data, outcome)) {
    f3 <- as.formula(paste(outcome, "~ ecoli * arsenic + AGE_M_CENTERED", sex_term,
                           "+", extra_str))
    m3 <- svyglm(f3, design = design, family = quasibinomial())
    interaction_test <- regTermTest(m3, ~ecoli:arsenic)
    m3_results <- or_table(m3, "Model 3 (interaction)", outcome)
  } else {
    cat("  [", outcome, "] Model 3 (interaction) SKIPPED: not estimable ",
        "(a ecoli x arsenic combination has zero children in one outcome ",
        "category)\n", sep = "")
    m3 <- NULL
    interaction_test <- NULL
    m3_results <- NULL
  }
  
  list(
    models = list(model0 = m0, model1 = m1, model2 = m2, model3 = m3),
    dropped_covariates = zero_cell_vars,
    results = bind_rows(
      or_table(m0, "Model 0", outcome),
      or_table(m1, "Model 1", outcome),
      or_table(m2, "Model 2", outcome),
      m3_results
    ),
    interaction_test = interaction_test
  )
}

## =============================================================================
## Gap 3: Does stunting mediate the E. coli -> ECDI relationship?
## Bangladesh MICS 2019
##
## Causal chain under test:
##      HH_DW_ECOLI (E. coli contamination, X)
##            |
##            v
##      Stunting / HAZ  (mediator, M)
##            |
##            v
##      ECDI_TRACK  (developmentally on-track, Y)
##
## Two parallel mediator specifications are tested, as recommended by the
## supervisor's framework:
##   (A) Binary mediator  : stunted (HAZ < -2 SD)      -> logistic mediator model
##   (B) Continuous mediator: Z_SCORE_HEIGHT (HAZ)      -> linear mediator model
##
## Design notes:
##   - MICS 2019 is a stratified, clustered, weighted survey (PSU, STRATUM, WEIGHT).
##   - The `mediation` package's mediate() does NOT accept svyglm objects directly,
##     so weighted glm()/lm() models are fit with `weights = WEIGHT`, and cluster-
##     robust inference is obtained via mediate(..., boot = TRUE, cluster = PSU).
##     NOTE: boot = TRUE is REQUIRED for the cluster argument to take effect --
##     boot = FALSE silently ignores clustering.
## =============================================================================

## ---- 0. Setup ---------------------------------------------------------------
required_pkgs <- c("readxl", "survey", "mediation", "dplyr", "boot", "car")
new_pkgs <- required_pkgs[!(required_pkgs %in% installed.packages()[, "Package"])]

extract_mediate_results <- function(med_obj, pathway_label) {

    ACME_p           = s$d.avg.p,

  )
}
